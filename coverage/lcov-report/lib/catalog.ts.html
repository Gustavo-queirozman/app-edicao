async function cached<T>(key: string, load: () => Promise<T>, fallback: T): Promise<T> { try { const response = await load(); await AsyncStorage.setItem(key, JSON.stringify({ savedAt: Date.now(), value: response } satisfies Cached<T>)); return response; } catch { const raw = await AsyncStorage.getItem(key); return raw ? (JSON.parse(raw) as Cached<T>).value : fallback; } }
