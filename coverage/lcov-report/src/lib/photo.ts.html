import * as ImagePicker from 'expo-image-picker';
import { manipulateAsync, SaveFormat } from 'expo-image-manipulator';
import { File } from 'expo-file-system';
 
const allowed = new Set(['image/jpeg', 'image/png', 'image/webp']);
export type PreparedImage = { uri: string; mimeType: 'image/jpeg' | 'image/png' | 'image/webp'; name: string };
function mimeFromMagic(bytes: Uint8Array) { const b = (index: number) => bytes[index]; if (b(0) === 0xff && b(1) === 0xd8) return 'image/jpeg'; if (b(0) === 0x89 && b(1) === 0x50 && b(2) === 0x4e && b(3) === 0x47) return 'image/png'; if (String.fromCharCode(...bytes.slice(0, 4)) === 'RIFF' && String.fromCharCode(...bytes.slice(8, 12)) === 'WEBP') return 'image/webp'; return undefined; }
export async function choosePhoto(fromCamera = false): Promise<PreparedImage | null> { const permission = fromCamera ? await ImagePicker.requestCameraPermissionsAsync() : await ImagePicker.requestMediaLibraryPermissionsAsync(); if (!permission.granted) throw new Error('Permissão necessária para escolher uma foto.'); const picked = fromCamera ? await ImagePicker.launchCameraAsync({ mediaTypes: ['images'], quality: 1 }) : await ImagePicker.launchImageLibraryAsync({ mediaTypes: ['images'], quality: 1 }); if (picked.canceled) return null; const asset = picked.assets[0]; const mimeType = mimeFromMagic((await new File(asset.uri).bytes()).slice(0, 16)); if (!mimeType || !allowed.has(mimeType)) throw new Error('Escolha uma imagem JPG, PNG ou WEBP válida.'); const resized = await manipulateAsync(asset.uri, [{ resize: asset.width > 2048 ? { width: 2048 } : {} }], { compress: .9, format: mimeType === 'image/png' ? SaveFormat.PNG : SaveFormat.JPEG, base64: false }); return { uri: resized.uri, mimeType: mimeType === 'image/png' ? 'image/png' : 'image/jpeg', name: `foto-${Date.now()}.${mimeType === 'image/png' ? 'png' : 'jpg'}` }; }
