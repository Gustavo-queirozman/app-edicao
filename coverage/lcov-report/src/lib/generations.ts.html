import * as Crypto from 'expo-crypto';
import { Directory, File, Paths } from 'expo-file-system';
import { request } from './http';
import { PreparedImage } from './photo';
import { Session, freshToken } from './session';
 
export type GenerationStatus = 'queued' | 'processing' | 'completed' | 'failed' | 'expired' | 'cancelled';
export type Generation = { id: string; status: GenerationStatus; errorCode?: 'openai_failed' | 'content_blocked' | 'insufficient_credits' | 'rate_limited'; resultChecksum?: string };
const auth = async (session: Session) => ({ Authorization: `Bearer ${(await freshToken(session)).accessToken}` });
export async function createGeneration(session: Session, effectId: string, image: PreparedImage, idempotencyKey = Crypto.randomUUID()) { const form = new FormData(); form.append('effectId', effectId); form.append('image', { uri: image.uri, type: image.mimeType, name: image.name } as unknown as Blob); return request<Generation>('/generations', { method: 'POST', headers: { ...(await auth(session)), 'Idempotency-Key': idempotencyKey }, body: form }); }
export async function getGeneration(session: Session, id: string) { return request<Generation>(`/generations/${id}`, { headers: await auth(session) }); }
export async function pollGeneration(session: Session, id: string, onChange: (generation: Generation) => void, signal?: AbortSignal) { for (let wait = 1000; !signal?.aborted; wait = Math.min(wait * 1.6, 10000)) { const generation = await getGeneration(session, id); onChange(generation); if (['completed', 'failed', 'expired', 'cancelled'].includes(generation.status)) return generation; await new Promise(resolve => setTimeout(resolve, wait)); } throw new Error('Acompanhamento cancelado.'); }
export async function downloadAndAcknowledge(session: Session, generation: Generation) { const directory = new Directory(Paths.document, 'creations'); directory.create({ idempotent: true, intermediates: true }); const temporary = new File(directory, `${generation.id}.tmp`); const result = await File.downloadFileAsync(`${(await import('./config')).config.apiUrl}/generations/${generation.id}/result`, temporary, { headers: await auth(session), idempotent: true }); const hash = await Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, await result.base64()); if (generation.resultChecksum && hash !== generation.resultChecksum) { result.delete(); throw new Error('O arquivo baixado não passou na verificação.'); } const saved = new File(directory, `${generation.id}.jpg`); if (saved.exists) saved.delete(); await result.move(saved); await request(`/generations/${generation.id}/ack`, { method: 'POST', headers: { ...(await auth(session)), 'Content-Type': 'application/json' }, body: JSON.stringify({ checksum: hash }) }); return saved.uri; }
