import { useState } from 'react';
import { Alert, Image, StyleSheet, Text, View } from 'react-native';
import { Button, Card } from './ui';
import { choosePhoto, PreparedImage } from '../lib/photo';
import { createGeneration, downloadAndAcknowledge, pollGeneration } from '../lib/generations';
import { saveCreation } from '../lib/database';
import { Session } from '../lib/session';
import { colors, spacing } from '../theme';
 
export function Composer({ session, onDone }: { session: Session; onDone: () => void }) {
  const [photo, setPhoto] = useState<PreparedImage | null>(null); const [busy, setBusy] = useState(false); const [status, setStatus] = useState('');
  const select = async (camera = false) => { try { setPhoto(await choosePhoto(camera)); } catch (error) { Alert.alert('Não foi possível usar esta foto', error instanceof Error ? error.message : 'Tente outra imagem.'); } };
  const generate = async () => { if (!photo) return; setBusy(true); try { setStatus('Enviando sua foto com segurança...'); const generation = await createGeneration(session, 'editorial', photo); await saveCreation({ id: generation.id, effectId: 'editorial', effectName: 'Retrato editorial', sourceUri: photo.uri, status: generation.status, createdAt: Date.now() }); const final = await pollGeneration(session, generation.id, g => setStatus(g.status === 'queued' ? 'Na fila...' : g.status === 'processing' ? 'A IA está criando sua edição...' : 'Finalizando...')); if (final.status !== 'completed') throw new Error(final.errorCode === 'insufficient_credits' ? 'Seu saldo é insuficiente.' : final.errorCode === 'content_blocked' ? 'Esta imagem não pôde ser processada.' : 'A geração não foi concluída.'); const resultUri = await downloadAndAcknowledge(session, final); await saveCreation({ id: final.id, effectId: 'editorial', effectName: 'Retrato editorial', sourceUri: photo.uri, resultUri, status: 'completed', createdAt: Date.now() }); Alert.alert('Edição pronta', 'A imagem foi salva com segurança neste dispositivo.'); onDone(); } catch (error) { Alert.alert('Geração não concluída', error instanceof Error ? error.message : 'Tente novamente.'); } finally { setBusy(false); setStatus(''); } };
  return <View style={styles.page}><Text style={styles.title}>Nova edição</Text><Text style={styles.copy}>Retrato editorial · 1 foto · 2 créditos</Text><Card style={styles.preview}>{photo ? <Image source={{ uri: photo.uri }} style={styles.image} /> : <Text style={styles.placeholder}>Escolha uma foto para pré-visualizar</Text>}</Card><Text style={styles.privacy}>Ao gerar, a foto será transmitida temporariamente para processamento por IA. Ela não é enviada ao nosso armazenamento de longo prazo.</Text>{status ? <Text style={styles.status}>{status}</Text> : null}<Button disabled={busy} label="Escolher da galeria" onPress={() => select(false)} /><Button disabled={busy} label="Usar câmera" tone="ghost" onPress={() => select(true)} />{photo ? <Button disabled={busy} label={busy ? 'Criando edição...' : 'Confirmar e gerar'} onPress={generate} /> : null}<Button disabled={busy} label="Cancelar" tone="ghost" onPress={onDone} /></View>;
}
const styles = StyleSheet.create({ page: { flex: 1, padding: spacing.lg, gap: spacing.md, backgroundColor: colors.canvas }, title: { color: colors.ink, fontWeight: '900', fontSize: 27 }, copy: { color: colors.muted }, preview: { height: 280, padding: 0, overflow: 'hidden', alignItems: 'center', justifyContent: 'center' }, image: { height: '100%', width: '100%', resizeMode: 'contain' }, placeholder: { color: colors.muted, padding: spacing.xl, textAlign: 'center' }, privacy: { color: colors.muted, fontSize: 12, lineHeight: 18 }, status: { color: colors.primary, textAlign: 'center', fontWeight: '800' } });
